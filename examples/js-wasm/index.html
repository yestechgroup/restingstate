Usage

import init, { JsMachine } from './pkg/xstate_wasm.js';
await init();

const machine = new JsMachine(JSON.stringify({
  id: "toggle", initial: "inactive",
  context: { fields: [] },
  states: [
    { name: "inactive", transitions: [{ event: "TOGGLE", target: "active" }],
      children: [], tags: [], entry: [], exit: [], on_done: [], output: null },
    { name: "active", transitions: [{ event: "TOGGLE", target: "inactive" }],
      children: [], tags: [], entry: [], exit: [], on_done: [], output: null }
  ]
}));

machine.send(JSON.stringify({ type: "TOGGLE" }));
console.log(JSON.parse(machine.snapshot()).value); // "active"
